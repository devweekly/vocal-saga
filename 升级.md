# Phase 1: Feature Graph Construction

👉 替代 walker 的 accept/grab

```ts id="k2v8q1"
function buildGraph(root: Element): Map<Element, NodeState> {
  const graph = new Map<Element, NodeState>();

  function visit(el: Element, depth: number) {
    if (hardReject(el)) return;

    const features = extractFeatures(el);

    graph.set(el, {
      features,
      baseScore: computeBaseScore(features),
      propagatedScore: 0,
      clusterId: -1,
    });

    const children = el.children;
    for (let i = 0; i < children.length; i++) {
      visit(children[i], depth + 1);
    }

    const shadow = (el as any).shadowRoot;
    if (shadow?.children) {
      for (let i = 0; i < shadow.children.length; i++) {
        visit(shadow.children[i], depth + 1);
      }
    }
  }

  visit(root, 0);

  return graph;
}
```

## ✔ propagation rule

```ts id="g8k3p9"
function propagateScores(graph: Map<Element, NodeState>) {
  for (const [el, node] of graph) {
    let boost = 0;

    // sibling reinforcement
    const siblings = el.parentElement?.children || [];
    let siblingTextMass = 0;

    for (let i = 0; i < siblings.length; i++) {
      const s = graph.get(siblings[i]);
      if (s) siblingTextMass += s.features.textLength;
    }

    if (siblingTextMass > 5000) {
      boost += 0.3; // cluster reinforcement
    }

    // parent context boost
    const parent = el.parentElement;
    if (parent && graph.has(parent)) {
      const p = graph.get(parent)!;
      if (p.features.semantic.isArticle) {
        boost += 0.4;
      }
    }

    // depth normalization
    const depthPenalty = 1 / (node.features.depth + 1);

    node.propagatedScore =
      node.baseScore * (1 + boost) * depthPenalty;
  }
}
```

## ✔ clustering algorithm

```ts id="x7nqk2"
function clusterBlocks(graph: Map<Element, NodeState>) {
  const visited = new Set<Element>();
  const clusters: Element[][] = [];

  for (const [el, node] of graph) {
    if (visited.has(el)) continue;

    if (node.propagatedScore < SCORE_THRESHOLD) continue;

    const cluster: Element[] = [];

    dfsCluster(el, cluster, visited, graph);

    if (cluster.length > 0) {
      clusters.push(cluster);
    }
  }

  return clusters;
}
```

## ✔ DFS clustering rule（关键）

```ts id="p9c0m4"
function dfsCluster(
  el: Element,
  cluster: Element[],
  visited: Set<Element>,
  graph: Map<Element, NodeState>
) {
  if (visited.has(el)) return;
  visited.add(el);

  const node = graph.get(el);
  if (!node || node.propagatedScore < SCORE_THRESHOLD) return;

  cluster.push(el);

  const siblings = el.parentElement?.children || [];
  for (let i = 0; i < siblings.length; i++) {
    dfsCluster(siblings[i], cluster, visited, graph);
  }
}
```

# Phase 4: Block Assembly（替代 grabNode）

```ts id="r2v7d1"
function buildBlocks(
  clusters: Element[],
  seenTexts: Set<string>,
  blockIdRef: { value: number }
): TextBlock[] {
  const blocks: TextBlock[] = [];

  for (const cluster of clusters) {
    const text = cluster
      .map(el => el.textContent || '')
      .join('\n')
      .trim();

    if (!text || seenTexts.has(text)) continue;

    seenTexts.add(text);

    const representative = cluster[0];

    blocks.push({
      id: `b${++blockIdRef.value}`,
      tag: representative.tagName.toLowerCase(),
      text,
      xpath: getXPath(representative),
      context: {},
    });
  }

  return blocks;
}
```
